class Player {
  constructor(name, id) {
    this.id = id;
    this.name = name;
    this.health = 100;
    this.maxHealth = 100;
    this.level = 1;
    this.experience = 0;
    this.inventory = [];
    this.position = { x: 0, y: 0 };
    this.isAlive = true;
  }

  takeDamage(amount) {
    if (!this.isAlive) return;
    
    this.health = Math.max(0, this.health - amount);
    console.log(`${this.name} took ${amount} damage. Health: ${this.health}/${this.maxHealth}`);
    
    if (this.health <= 0) {
      this.die();
    }
  }

  heal(amount) {
    if (!this.isAlive) return;
    
    this.health = Math.min(this.maxHealth, this.health + amount);
    console.log(`${this.name} healed ${amount} HP. Health: ${this.health}/${this.maxHealth}`);
  }

  gainExperience(amount) {
    this.experience += amount;
    console.log(`${this.name} gained ${amount} XP. Total: ${this.experience}`);
    
    // Simple level up logic
    const xpForNextLevel = this.level * 100;
    if (this.experience >= xpForNextLevel) {
      this.levelUp();
    }
  }

  levelUp() {
    this.level++;
    this.experience = 0;
    this.maxHealth += 20;
    this.health = this.maxHealth;
    console.log(`🎉 ${this.name} leveled up to level ${this.level}! Max health increased to ${this.maxHealth}.`);
  }

  die() {
    this.isAlive = false;
    this.health = 0;
    console.log(`💀 ${this.name} has died!`);
    // Trigger death event, respawn logic, etc.
  }

  move(x, y) {
    this.position.x = x;
    this.position.y = y;
    console.log(`${this.name} moved to (${x}, ${y})`);
  }

  addItem(item) {
    this.inventory.push(item);
    console.log(`${this.name} picked up: ${item.name}`);
  }

  removeItem(itemName) {
    const index = this.inventory.findIndex(item => item.name === itemName);
    if (index !== -1) {
      const removed = this.inventory.splice(index, 1)[0];
      console.log(`${this.name} dropped: ${removed.name}`);
      return removed;
    }
    return null;
  }

  getStatus() {
    return {
      id: this.id,
      name: this.name,
      health: this.health,
      maxHealth: this.maxHealth,
      level: this.level,
      experience: this.experience,
      position: this.position,
      isAlive: this.isAlive,
      inventory: this.inventory.map(item => item.name)
    };
  }
}
